    <script>
        // รับค่ามาจาก doGet (ระบบหลัก)
        const SYSTEM_USER_ID = "<?= typeof uid !== 'undefined' ? uid : 'guest' ?>";

        let SHOW_RAW_ASSIGNMENTS = false; 
        let MANUAL_SCORES = {}; 

        const HISTORY_MANAGER = {
            past: [],
            future: [],
            
            saveState: function() {
                this.past.push(JSON.parse(JSON.stringify(MANUAL_SCORES)));
                if(this.past.length > 50) this.past.shift(); 
                this.future = []; 
                this.updateButtons();
            },
            
            undo: function() {
                if(this.past.length === 0) return;
                this.future.push(JSON.parse(JSON.stringify(MANUAL_SCORES)));
                MANUAL_SCORES = this.past.pop();
                this.updateButtons();
                calculateAll();
            },
            
            redo: function() {
                if(this.future.length === 0) return;
                this.past.push(JSON.parse(JSON.stringify(MANUAL_SCORES)));
                MANUAL_SCORES = this.future.pop();
                this.updateButtons();
                calculateAll();
            },
            
            updateButtons: function() {
                const btnUndo = document.getElementById('btn-undo');
                const btnRedo = document.getElementById('btn-redo');
                
                if(btnUndo) {
                    btnUndo.disabled = this.past.length === 0;
                    if(!btnUndo.disabled) { btnUndo.classList.add('bg-blue-100', 'text-blue-600', 'border-blue-200'); btnUndo.classList.remove('bg-slate-100', 'text-slate-400'); }
                    else { btnUndo.classList.remove('bg-blue-100', 'text-blue-600', 'border-blue-200'); btnUndo.classList.add('bg-slate-100', 'text-slate-400'); }
                }
                
                if(btnRedo) {
                    btnRedo.disabled = this.future.length === 0;
                    if(!btnRedo.disabled) { btnRedo.classList.add('bg-green-100', 'text-green-600', 'border-green-200'); btnRedo.classList.remove('bg-slate-100', 'text-slate-400'); }
                    else { btnRedo.classList.remove('bg-green-100', 'text-green-600', 'border-green-200'); btnRedo.classList.add('bg-slate-100', 'text-slate-400'); }
                }
            }
        };

        window.onload = () => { getInitialDropdowns(); };

        function showLoading(show, text = 'กำลังโหลดข้อมูล...') {

            if (!overlay) return;

            }
        }

        function getInitialDropdowns() {
            showLoading(true, 'กำลังดึงข้อมูลระบบ...');
            google.script.run

        function openClearDataModal() {

        function confirmClearData() {
            const chkPre = document.getElementById('chk-clear-pre').checked;
            const chkMid = document.getElementById('chk-clear-mid').checked;
            const chkPost = document.getElementById('chk-clear-post').checked;
            const chkFinal = document.getElementById('chk-clear-final').checked;

            if (!chkPre && !chkMid && !chkPost && !chkFinal) {
                return Swal.fire('เตือน', 'กรุณาเลือกอย่างน้อย 1 ช่องที่ต้องการล้างข้อมูล', 'warning');
            }

            Swal.fire({
                title: 'ยืนยันการล้างข้อมูล?',
                text: 'การเปลี่ยนแปลงจะแสดงบนตาราง (ต้องกดบันทึกเพื่อเซฟลงระบบ)',
                icon: 'warning',
                showCancelButton: true,
                confirmButtonColor: '#d33',
                cancelButtonColor: '#94a3b8',
                confirmButtonText: 'ล้างข้อมูลเลย'
            }).then((result) => {
                if (result.isConfirmed) {
                    HISTORY_MANAGER.saveState();
                    
                    GRADE_DATA.students.forEach(std => {
                        const sid = std.id;
                        if (!MANUAL_SCORES[sid]) MANUAL_SCORES[sid] = {};
                        
                        let oldPre = MANUAL_SCORES[sid]['pre'] !== undefined ? MANUAL_SCORES[sid]['pre'] : (GRADE_DATA.exams[sid] ? GRADE_DATA.exams[sid].pre : '');
                        let oldMid = MANUAL_SCORES[sid]['mid'] !== undefined ? MANUAL_SCORES[sid]['mid'] : (GRADE_DATA.exams[sid] ? GRADE_DATA.exams[sid].mid : '');
                        let oldPost = MANUAL_SCORES[sid]['post'] !== undefined ? MANUAL_SCORES[sid]['post'] : (GRADE_DATA.exams[sid] ? GRADE_DATA.exams[sid].post : '');
                        let oldFinal = MANUAL_SCORES[sid]['final'] !== undefined ? MANUAL_SCORES[sid]['final'] : (GRADE_DATA.exams[sid] ? GRADE_DATA.exams[sid].final : '');

                        const tArray = window.CURRENT_TEACHER || [];
                        const tName = tArray.length > 0 ? tArray[0].teacherName : 'ครูผู้สอน';
                        const nowStamp = new Date().toLocaleTimeString('th-TH');

                        const applyClear = (type, oldVal) => {
                            MANUAL_SCORES[sid][type] = '';
                            if (String(oldVal) !== '') {
                                if (!GRADE_DATA.adjustments[sid]) GRADE_DATA.adjustments[sid] = [];
                                GRADE_DATA.adjustments[sid].push({
                                    id: 'LOCAL-' + Date.now() + Math.floor(Math.random()*100),
                                    timestamp: nowStamp, by: tName, type: type,
                                    oldValue: oldVal, newValue: '-', reason: 'ใช้เมนูล้างข้อมูล'
                                });
                            }
                            const inp = document.getElementById(`${type}-${sid}`);
                            if (inp) inp.value = '';
                        };

                        if (chkPre) applyClear('pre', oldPre);
                        if (chkMid) applyClear('mid', oldMid);
                        if (chkPost) applyClear('post', oldPost);
                        if (chkFinal) applyClear('final', oldFinal);
                        
                        updateRowCalculations(sid);
                    });

        function handleCellChange(sid, type, value, inputEl) {
            HISTORY_MANAGER.saveState();
            
            const numVal = value === '' ? '' : Number(value);
            if (!MANUAL_SCORES[sid]) MANUAL_SCORES[sid] = {};
            
            let oldVal = MANUAL_SCORES[sid][type];
            if(oldVal === undefined) {
                if (type === 'pre') oldVal = GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].pre !== '' ? GRADE_DATA.exams[sid].pre : '';
                else if (type === 'mid') oldVal = GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].mid !== '' ? GRADE_DATA.exams[sid].mid : '';
                else if (type === 'post') oldVal = GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].post !== '' ? GRADE_DATA.exams[sid].post : '';
                else if (type === 'final') oldVal = GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].final !== '' ? GRADE_DATA.exams[sid].final : '';
            }

            MANUAL_SCORES[sid][type] = numVal;

            if (String(numVal) !== String(oldVal)) {
                const tArray = window.CURRENT_TEACHER || [];
                const tName = tArray.length > 0 ? tArray[0].teacherName : 'ครูผู้สอน';
                if (!GRADE_DATA.adjustments[sid]) GRADE_DATA.adjustments[sid] = [];
                
                const nowStamp = new Date().toLocaleTimeString('th-TH');
                const existingSessionLog = GRADE_DATA.adjustments[sid].find(a => a.type === type && a.id.startsWith('LOCAL-'));
                
                if (existingSessionLog) {
                    existingSessionLog.newValue = numVal;
                    existingSessionLog.timestamp = nowStamp;
                } else {
                    GRADE_DATA.adjustments[sid].push({
                        id: 'LOCAL-' + Date.now() + Math.floor(Math.random()*100),
                        timestamp: nowStamp,
                        by: tName,
                        type: type,
                        oldValue: oldVal !== '' ? oldVal : '-',
                        newValue: numVal,
                        reason: 'พิมพ์แก้ไขในตาราง'
                    });
                }
            }

            updateRowCalculations(sid);
            recalculateStats();
        }

        function initExcelBehavior() {
            const table = document.getElementById('grading-table');
            if(!table) return;

            const clearFocus = () => {
                document.querySelectorAll('.cell-focused').forEach(td => td.classList.remove('cell-focused'));
            };

            table.addEventListener('click', function(e) {
                if(e.target.classList.contains('excel-input')) {
                    clearFocus();
                    e.target.parentElement.classList.add('cell-focused');
                    e.target.select();
                }
            });

            table.addEventListener('focusin', function(e) {
                if(e.target.classList.contains('excel-input')) {
                    clearFocus();
                    e.target.parentElement.classList.add('cell-focused');
                    e.target.select();
                }
            });

            table.addEventListener('keydown', function(e) {
                if(!e.target.classList.contains('excel-input')) return;
                
                const currentTd = e.target.parentElement;
                const currentTr = currentTd.parentElement;
                const allInputs = Array.from(currentTr.querySelectorAll('.excel-input'));
                const colIndex = allInputs.indexOf(e.target);
                
                let nextInput = null;

                switch(e.key) {
                    case 'ArrowDown':
                    case 'Enter':
                        e.preventDefault();
                        const nextTrDown = currentTr.nextElementSibling;
                        if(nextTrDown) nextInput = Array.from(nextTrDown.querySelectorAll('.excel-input'))[colIndex];
                        break;
                    case 'ArrowUp':
                        e.preventDefault();
                        const prevTrUp = currentTr.previousElementSibling;
                        if(prevTrUp) nextInput = Array.from(prevTrUp.querySelectorAll('.excel-input'))[colIndex];
                        break;
                    case 'ArrowRight':
                        if(e.target.selectionStart === e.target.value.length || e.target.value === '') {
                            e.preventDefault();
                            if (colIndex < allInputs.length - 1) nextInput = allInputs[colIndex + 1];
                        }
                        break;
                    case 'ArrowLeft':
                        if(e.target.selectionEnd === 0 || e.target.value === '') {
                            e.preventDefault();
                            if (colIndex > 0) nextInput = allInputs[colIndex - 1];
                        }
                        break;
                }

                if(nextInput) {
                    clearFocus();
                    nextInput.parentElement.classList.add('cell-focused');
                    nextInput.focus();
                    nextInput.select();
                }
            });

            table.addEventListener('paste', function(e) {
                if(!e.target.classList.contains('excel-input')) return;
                e.preventDefault();
                
                const clipboardData = e.clipboardData || window.clipboardData;
                const pastedData = clipboardData.getData('Text');
                
                const rawRows = pastedData.split('\n');
                const validRows = rawRows.filter(r => r.trim() !== '');
                
                const startTd = e.target.parentElement;
                let currentTr = startTd.parentElement;
                const allInputs = Array.from(currentTr.querySelectorAll('.excel-input'));
                const startColIndex = allInputs.indexOf(e.target);
                
                const totalStudents = GRADE_DATA.students.length;

                if (validRows.length > 5 && validRows.length !== totalStudents) {
                    Swal.fire({
                        title: 'จำนวนข้อมูลไม่ตรงกัน!',
                        html: `<div class="text-left text-sm mt-2">
                                <p class="text-red-500 font-bold mb-1"><i class="fa-solid fa-triangle-exclamation"></i> คุณกำลังวางข้อมูล <b>${validRows.length}</b> แถว</p>
                                <p class="text-slate-600 mb-3">แต่นักเรียนในห้องนี้มีทั้งหมด <b>${totalStudents}</b> คน</p>
                                <p class="text-xs text-slate-500">อาจเกิดจากการคัดลอกห้องผิด หรือมีเด็กย้าย/ลาออก<br>ต้องการให้ระบบบังคับวางข้อมูลต่อไปหรือไม่?</p>
                               </div>`,
                        icon: 'warning',
                        showCancelButton: true,
                        confirmButtonColor: '#4e73df',
                        cancelButtonColor: '#94a3b8',
                        confirmButtonText: 'ยืนยัน วางข้อมูล',
                        cancelButtonText: 'ยกเลิก'
                    }).then((result) => {
                        if (result.isConfirmed) { processPaste(validRows, currentTr, startColIndex); }
                    });
                } else {
                    processPaste(validRows, currentTr, startColIndex);
                }
                
                function processPaste(rowsData, startTr, colIndex) {
                    let iterTr = startTr;
                    let errorCount = 0;

                    for(let r = 0; r < rowsData.length; r++) {
                        if(!iterTr) break; 
                        const cols = rowsData[r].split('\t'); 
                        const inputsInRow = Array.from(iterTr.querySelectorAll('.excel-input'));
                        
                        for(let c = 0; c < cols.length; c++) {
                            const targetInput = inputsInRow[colIndex + c];
                            if(targetInput) {
                                const val = cols[c].trim();
                                if(val !== '') {
                                    if(isNaN(val)) {
                                        errorCount++;
                                    } else {
                                        targetInput.value = val;
                                        handleCellChange(targetInput.id.split('-')[1], targetInput.id.split('-')[0], val, targetInput);
                                    }
                                }
                            }
                        }
                        iterTr = iterTr.nextElementSibling;
                    }
                    
                    if (errorCount > 0) {
                        Swal.fire({
                            toast: true, position: 'top-end',
                            icon: 'warning', title: `ข้ามข้อมูลที่ไม่ใช่ตัวเลข ${errorCount} ช่อง`,
                            showConfirmButton: false, timer: 3000
                        });
                    }
                }
            });
        }

        function updateRowCalculations(sid) {
            const preVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['pre'] !== undefined ? MANUAL_SCORES[sid]['pre'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].pre !== '' ? GRADE_DATA.exams[sid].pre : '');
            const midVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['mid'] !== undefined ? MANUAL_SCORES[sid]['mid'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].mid !== '' ? GRADE_DATA.exams[sid].mid : '');
            const postVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['post'] !== undefined ? MANUAL_SCORES[sid]['post'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].post !== '' ? GRADE_DATA.exams[sid].post : '');
            const finalVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['final'] !== undefined ? MANUAL_SCORES[sid]['final'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].final !== '' ? GRADE_DATA.exams[sid].final : '');

            const totalScore = (Number(preVal)||0) + (Number(midVal)||0) + (Number(postVal)||0) + (Number(finalVal)||0);

            let grade = "0";
            if(totalScore >= 80) grade = "4";
            else if(totalScore >= 75) grade = "3.5";
            else if(totalScore >= 70) grade = "3";
            else if(totalScore >= 65) grade = "2.5";
            else if(totalScore >= 60) grade = "2";
            else if(totalScore >= 55) grade = "1.5";
            else if(totalScore >= 50) grade = "1";

            let attScore = "1";
            if(totalScore >= 65) attScore = "3";
            else if(totalScore >= 55) attScore = "2";
            else if(totalScore >= 50) attScore = "1";

            let readScore = "1";
            if(totalScore >= 70) readScore = "3";
            else if(totalScore >= 55) readScore = "2";
            else if(totalScore >= 50) readScore = "1";

            window.SGS_EXPORT_DATA[sid] = {
                pre: preVal, mid: midVal, post: postVal, final: finalVal, total: totalScore, grade: grade,
                att: Array(8).fill(attScore),
                read: Array(5).fill(readScore),
                attTotal: attScore,
                readTotal: readScore
            };

            const tr = document.getElementById(`tr-${sid}`);
            if (tr) {
                const totalTd = tr.querySelector('.total-cell');
                const gradeTd = tr.querySelector('.grade-cell');
                const attTd = tr.querySelector('.att-cell');
                const readTd = tr.querySelector('.read-cell');
                const btnAudit = tr.querySelector('.btn-audit');
                
                const inpPre = tr.querySelector(`#pre-${sid}`);
                const inpMid = tr.querySelector(`#mid-${sid}`);
                const inpPost = tr.querySelector(`#post-${sid}`);
                const inpFinal = tr.querySelector(`#final-${sid}`);

                const checkWarn = (val, max, isMandatory, inpEl) => {
                    if (!inpEl) return;
                    inpEl.parentElement.classList.remove('warn-score', 'warn-score-final');
                    if (val !== '' && Number(val) < (max / 2)) {
                        if(isMandatory) inpEl.parentElement.classList.add('warn-score');
                        else inpEl.parentElement.classList.add('warn-score-final');
                    }
                };
                checkWarn(preVal, CURRENT_WEIGHTS.pre, true, inpPre);
                checkWarn(midVal, CURRENT_WEIGHTS.mid, true, inpMid);
                checkWarn(postVal, CURRENT_WEIGHTS.post, true, inpPost);
                checkWarn(finalVal, CURRENT_WEIGHTS.final, false, inpFinal);

                if(totalTd) {
                    totalTd.innerText = totalScore;
                    const modTotal = totalScore % 10;
                    if ([3, 4, 8, 9].includes(modTotal) && totalScore > 0) {
                        totalTd.classList.add('warn-total');
                    } else {
                        totalTd.classList.remove('warn-total');
                    }
                }
                
                if(gradeTd) {
                    gradeTd.innerText = grade;
                    gradeTd.className = `p-2 text-center border-r border-slate-100 bg-green-50 font-bold text-xl shadow-sm grade-cell ${grade=='0'?'text-red-600':'text-green-700'}`;
                }
                if(attTd) attTd.innerText = attScore;
                if(readTd) readTd.innerText = readScore;

                const hasAdjustments = GRADE_DATA.adjustments && GRADE_DATA.adjustments[sid] && GRADE_DATA.adjustments[sid].length > 0;
                if(btnAudit) {
                    btnAudit.className = `btn-audit ${hasAdjustments ? 'text-orange-500 bg-orange-50 hover:bg-orange-100 border border-orange-200' : 'text-slate-400 hover:text-blue-600 bg-slate-100 hover:bg-blue-50 border border-transparent'} px-3 py-1.5 rounded-lg text-xs font-bold transition w-full shadow-sm flex items-center justify-center gap-1`;
                    btnAudit.innerHTML = `<i class="fa-solid fa-clock-rotate-left ${hasAdjustments ? 'animate-pulse' : ''}"></i> ${hasAdjustments ? 'ประวัติ' : 'ดูประวัติ'}`;
                }
            }
        }

        function recalculateStats() {
            let stats = { '4':0, '3.5':0, '3':0, '2.5':0, '2':0, '1.5':0, '1':0, '0':0 };
            let studentsOver3 = 0;

            GRADE_DATA.students.forEach(std => {
                const sid = std.id;
                const preVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['pre'] !== undefined ? MANUAL_SCORES[sid]['pre'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].pre !== '' ? GRADE_DATA.exams[sid].pre : '');
                const midVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['mid'] !== undefined ? MANUAL_SCORES[sid]['mid'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].mid !== '' ? GRADE_DATA.exams[sid].mid : '');
                const postVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['post'] !== undefined ? MANUAL_SCORES[sid]['post'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].post !== '' ? GRADE_DATA.exams[sid].post : '');
                const finalVal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['final'] !== undefined ? MANUAL_SCORES[sid]['final'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].final !== '' ? GRADE_DATA.exams[sid].final : '');

                const totalScore = (Number(preVal)||0) + (Number(midVal)||0) + (Number(postVal)||0) + (Number(finalVal)||0);

                let grade = "0";
                if(totalScore >= 80) grade = "4";
                else if(totalScore >= 75) grade = "3.5";
                else if(totalScore >= 70) grade = "3";
                else if(totalScore >= 65) grade = "2.5";
                else if(totalScore >= 60) grade = "2";
                else if(totalScore >= 55) grade = "1.5";
                else if(totalScore >= 50) grade = "1";

                if(Number(grade) >= 3) studentsOver3++;
                stats[grade]++;
            });

            ['4','35','3','25','2','15','1','0'].forEach(g => {
                const key = g.replace('35','3.5').replace('25','2.5').replace('15','1.5');
                const statEl = document.getElementById(`stat-${g}`);
                if (statEl) statEl.innerText = stats[key];
            });

            const goalPercent = CURRENT_WEIGHTS.goal;
            const currentPercent = GRADE_DATA.students.length > 0 ? Math.round((studentsOver3 / GRADE_DATA.students.length) * 100) : 0;
            const overallPercent = GRADE_DATA.overallTargetPercent || 0; 
            
            const currentText = document.getElementById('goal-current-text');
            const bar = document.getElementById('goal-progress-bar');
            const marker = document.getElementById('goal-marker');
            const dispGoalEl = document.getElementById('disp-goal');
            
            if (dispGoalEl) dispGoalEl.innerText = goalPercent;
            if (currentText) currentText.innerText = `${currentPercent}%`;
            if (bar) bar.style.width = `${currentPercent}%`;
            if (marker) marker.style.left = `${goalPercent}%`;

            if (bar && currentText) {
                if (currentPercent >= goalPercent) {
                    bar.classList.remove('bg-orange-400');
                    bar.classList.add('bg-green-500');
                    currentText.classList.add('text-green-600');
                } else {
                    bar.classList.remove('bg-green-500');
                    bar.classList.add('bg-orange-400');
                    currentText.classList.remove('text-green-600');
                }
            }

            const ovCurrentText = document.getElementById('overall-current-text');
            const ovBar = document.getElementById('overall-progress-bar');
            const ovMarker = document.getElementById('overall-goal-marker');
            
            if (ovCurrentText) ovCurrentText.innerText = `${overallPercent}%`;
            if (ovBar) ovBar.style.width = `${overallPercent}%`;
            if (ovMarker) ovMarker.style.left = `${goalPercent}%`;

            if (ovBar && ovCurrentText) {
                if (overallPercent >= goalPercent) {
                    ovBar.classList.remove('bg-indigo-400');
                    ovBar.classList.add('bg-green-500');
                    ovCurrentText.classList.add('text-green-600');
                    ovCurrentText.classList.remove('text-indigo-600');
                } else {
                    ovBar.classList.remove('bg-green-500');
                    ovBar.classList.add('bg-indigo-400');
                    ovCurrentText.classList.remove('text-green-600');
                    ovCurrentText.classList.add('text-indigo-600');
                }
            }
        }

        function collectTableData() {
            const tableData = [];
            GRADE_DATA.students.forEach(std => {
                const sid = std.id;
                const cPre = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['pre'] !== undefined ? MANUAL_SCORES[sid]['pre'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].pre !== '' ? GRADE_DATA.exams[sid].pre : '');
                const cMid = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['mid'] !== undefined ? MANUAL_SCORES[sid]['mid'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].mid !== '' ? GRADE_DATA.exams[sid].mid : '');
                const cPost = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['post'] !== undefined ? MANUAL_SCORES[sid]['post'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].post !== '' ? GRADE_DATA.exams[sid].post : '');
                const cFinal = MANUAL_SCORES[sid] && MANUAL_SCORES[sid]['final'] !== undefined ? MANUAL_SCORES[sid]['final'] : (GRADE_DATA.exams[sid] && GRADE_DATA.exams[sid].final !== '' ? GRADE_DATA.exams[sid].final : '');
                const total = (Number(cPre)||0) + (Number(cMid)||0) + (Number(cPost)||0) + (Number(cFinal)||0);

                let grade = "0";
                if(total >= 80) grade = "4"; else if(total >= 75) grade = "3.5"; else if(total >= 70) grade = "3";
                else if(total >= 65) grade = "2.5"; else if(total >= 60) grade = "2"; else if(total >= 55) grade = "1.5";
                else if(total >= 50) grade = "1";

                let attScore = "1"; if(total >= 65) attScore = "3"; else if(total >= 55) attScore = "2"; else if(total >= 50) attScore = "1";
                let readScore = "1"; if(total >= 70) readScore = "3"; else if(total >= 55) readScore = "2"; else if(total >= 50) readScore = "1";

                tableData.push({
                    no: std.no, id: sid, name: `${std.prefix}${std.first} ${std.last}`,
                    pre: cPre, mid: cMid, post: cPost, final: cFinal, total: total, grade: grade, att: attScore, read: readScore
                });
            });
            return tableData;
        }

        function saveAllData() {
            const term = document.getElementById('sel-term')?.value;
            const subject = document.getElementById('sel-subject')?.value;
            const level = document.getElementById('sel-level')?.value;
            const room = document.getElementById('sel-room')?.value;

            if (!subject || !level || !room) {
                return Swal.fire('เตือน', 'กรุณาโหลดข้อมูลห้องก่อนบันทึก', 'warning');
            }

            const batchPayloads = [];
            const localAdjustments = []; // เก็บประวัติการแก้ไขไปเซฟด้วย

            GRADE_DATA.students.forEach(std => {
                const sid = std.id;
                if (MANUAL_SCORES[sid]) {
                    let scores = {};
                    if (MANUAL_SCORES[sid]['pre'] !== undefined) scores.pre = MANUAL_SCORES[sid]['pre'];
                    if (MANUAL_SCORES[sid]['mid'] !== undefined) scores.mid = MANUAL_SCORES[sid]['mid'];
                    if (MANUAL_SCORES[sid]['post'] !== undefined) scores.post = MANUAL_SCORES[sid]['post'];
                    if (MANUAL_SCORES[sid]['final'] !== undefined) scores.final = MANUAL_SCORES[sid]['final'];
                    
                    if (Object.keys(scores).length > 0) {
                        batchPayloads.push({ term, subject, studentId: sid, scores: scores });
                    }
                }

                // ดึงประวัติที่ยังไม่ได้เซฟ (LOCAL)
                if (GRADE_DATA.adjustments[sid]) {
                    GRADE_DATA.adjustments[sid].forEach(adj => {
                        if (String(adj.id).startsWith('LOCAL-')) {
                            localAdjustments.push({
                                term: term,
                                subject: subject,
                                studentId: sid,
                                type: adj.type,
                                oldValue: adj.oldValue !== undefined ? adj.oldValue : '-',
                                newValue: adj.newValue !== undefined ? adj.newValue : (adj.newValue !== '-' ? Number(adj.newValue) : 0),
                                reason: adj.reason,
                                recorder: adj.by
                            });
                        }
                    });
                }
            });

            if (batchPayloads.length === 0 && localAdjustments.length === 0) {
                 return Swal.fire('ข้อมูล', 'ยังไม่มีการแก้ไขคะแนนใหม่ให้บันทึก', 'info');
            }

            Swal.fire({ title: 'กำลังบันทึกคะแนนเข้าระบบ...', allowOutsideClick: false, didOpen: () => { Swal.showLoading(); } });

            google.script.run.withSuccessHandler((res) => {
                if(res.success) {
                    const updateEl = document.getElementById('last-update-time');
                    if (updateEl) updateEl.innerText = new Date().toLocaleTimeString('th-TH') + ' น.';
                    MANUAL_SCORES = {};
                    
                    // เปลี่ยยสถานะเป็นบันทึกแล้ว จะได้ไม่เซฟซ้ำ
                    GRADE_DATA.students.forEach(std => {
                        if (GRADE_DATA.adjustments[std.id]) {
                            GRADE_DATA.adjustments[std.id].forEach(adj => {
                                if (String(adj.id).startsWith('LOCAL-')) {
                                    adj.id = 'SAVED-' + Date.now();
                                }
                            });
                        }
                    });

                    Swal.fire({ icon: 'success', title: 'บันทึกสำเร็จ', text: 'บันทึกคะแนนและประวัติเรียบร้อย', timer: 1500, showConfirmButton: false });
                } else {
                    Swal.fire('Error', res.message, 'error');
                }
            }).saveBatchExamScores(batchPayloads, localAdjustments, `${level}/${room}`, SYSTEM_USER_ID);
        }

        function checkTeacherWeight() {
            const pre = Number(document.getElementById('t-w-pre').value) || 0;
            const mid = Number(document.getElementById('t-w-mid').value) || 0;
            const post = Number(document.getElementById('t-w-post').value) || 0;
            const final = Number(document.getElementById('t-w-final').value) || 0;
            const total = pre + mid + post + final;
            
            const wStatus = document.getElementById('t-weight-status');
            if (total === 100) {
                wStatus.className = "text-[10px] bg-green-100 text-green-700 px-2 py-1 rounded-md border border-green-200";
                wStatus.innerText = "รวม 100 พอดี";
            } else {
                wStatus.className = "text-[10px] bg-red-100 text-red-700 px-2 py-1 rounded-md border border-red-200 animate-pulse";
                wStatus.innerText = `รวมได้ ${total}`;
            }
        }

        function openTeacherModal() {
            const tArray = window.CURRENT_TEACHER || [];
            const name = tArray.length > 0 ? tArray[0].teacherName : SYSTEM_USER_NAME;
            document.getElementById('t-name').value = name !== 'ไม่ระบุตัวตน' ? name : '';
            
            const selSubj = document.getElementById('t-subject-select');
            if (selSubj && window.SYSTEM_DATA && window.SYSTEM_DATA.subjects) {
                selSubj.innerHTML = '<option value="">-- เลือกรายวิชา --</option>' + 
                    window.SYSTEM_DATA.subjects.map(s => `<option value="${s.code}">${s.code} ${s.name}</option>`).join('');
            }
            
            document.getElementById('t-room-select').innerHTML = '<option value="">-- เลือกวิชาด้านซ้ายก่อน --</option>';
            checkTeacherWeight();
            document.getElementById('teacher-modal').classList.remove('hidden');
        }

        function onTeacherSubjectChange() {
            const subj = document.getElementById('t-subject-select').value;
            const roomSel = document.getElementById('t-room-select');
            
            if (!subj) {
                roomSel.innerHTML = '<option value="">-- เลือกวิชาด้านซ้ายก่อน --</option>';
                return;
            }

            if (window.SYSTEM_DATA && window.SYSTEM_DATA.levels && window.SYSTEM_DATA.rooms) {
                let options = '<option value="">-- เลือกห้องที่ต้องการตั้งค่า --</option>';
                window.SYSTEM_DATA.levels.forEach(l => {
                    window.SYSTEM_DATA.rooms.forEach(r => {
                        options += `<option value="${l}/${r}">${l}/${r}</option>`;
                    });
                });
                roomSel.innerHTML = options;
            }
        }

        function onTeacherRoomChange() {
             const term = document.getElementById('sel-term')?.value;
             const subj = document.getElementById('t-subject-select').value;
             const roomStr = document.getElementById('t-room-select').value;
             if(!subj || !roomStr) return;

             const [level, room] = roomStr.split('/');
             const filter = { term, subject: subj, level: level, room: room };
             
             google.script.run.withSuccessHandler(res => {
                 if (res.success && res.data.settings) {
                     const s = res.data.settings;
                     if(Object.keys(s).length > 0) {
                         document.getElementById('t-w-pre').value = s.weightPre != null ? s.weightPre : 20;
                         document.getElementById('t-w-mid').value = s.weightMid != null ? s.weightMid : 20;
                         document.getElementById('t-w-post').value = s.weightPost != null ? s.weightPost : 20;
                         document.getElementById('t-w-final').value = s.weightFinal != null ? s.weightFinal : 40;
                         document.getElementById('t-goal').value = s.goalPercent != null ? s.goalPercent : 66;
                         checkTeacherWeight();
                     }
                 }
             }).getGradingData(filter);
        }

        function saveTeacherAndSettingsData(e) {
            e.preventDefault();
            const term = document.getElementById('sel-term')?.value;
            const subj = document.getElementById('t-subject-select').value;
            const roomStr = document.getElementById('t-room-select').value;
            const teacherName = document.getElementById('t-name').value;

            if(!subj) return Swal.fire('เตือน', 'กรุณาเลือกรายวิชา', 'warning');
            if(!roomStr) return Swal.fire('เตือน', 'กรุณาเลือกห้องเรียน', 'warning');
            if(SYSTEM_USER_ID === 'guest') return Swal.fire('เตือน', 'กรุณาเข้าสู่ระบบผ่านระบบหลักก่อนทำการตั้งค่า (ไม่อนุญาตให้บุคคลทั่วไปบันทึก)', 'error');

            const total = Number(document.getElementById('t-w-pre').value) + Number(document.getElementById('t-w-mid').value) + Number(document.getElementById('t-w-post').value) + Number(document.getElementById('t-w-final').value);
            if (total !== 100) return Swal.fire('เตือน', 'สัดส่วนคะแนนรวมต้องเท่ากับ 100 พอดี', 'warning');

            const btn = document.getElementById('btn-t-save');
            btn.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> บันทึก...';
            btn.disabled = true;

            const [level, room] = roomStr.split('/');
            const tPayload = { teacherName: teacherName, subjectCode: subj, room: roomStr };

            const sPayload = {
                term: term, subject: subj, level: level, room: room,
                wPre: document.getElementById('t-w-pre').value,
                wMid: document.getElementById('t-w-mid').value,
                wPost: document.getElementById('t-w-post').value,
                wFinal: document.getElementById('t-w-final').value,
                goal: document.getElementById('t-goal').value
            };

            google.script.run.withSuccessHandler(res => {
                if (res.success) {
                    let tArray = window.CURRENT_TEACHER || [];
                    const idx = tArray.findIndex(t => String(t.subjectCode).trim() === String(subj).trim() && String(t.room).trim() === String(roomStr).trim());
                    if(idx > -1) tArray[idx] = tPayload;
                    else tArray.push(tPayload);
                    window.CURRENT_TEACHER = tArray;
                    
                    google.script.run.withSuccessHandler(res2 => {
                         btn.innerHTML = 'บันทึกการตั้งค่า'; btn.disabled = false;
                         closeModal('teacher-modal');
                         Swal.fire({icon: 'success', title: 'บันทึกสำเร็จ', timer: 1500, showConfirmButton: false});
                         
                         if (document.getElementById('sel-subject').value === subj && 
                             document.getElementById('sel-level').value === level && 
                             document.getElementById('sel-room').value === room) {
                             loadGradingData();
                         }
                    }).saveSettings(sPayload, SYSTEM_USER_ID);
                } else {
                    btn.innerHTML = 'บันทึกการตั้งค่า'; btn.disabled = false;
                    Swal.fire('Error', res.message, 'error');
                }
            }).saveTeacherSettings(tPayload, SYSTEM_USER_ID);
        }

        function copyTable(tableId) {
            const table = document.getElementById(tableId);
            const range = document.createRange();
            range.selectNode(table);
            window.getSelection().removeAllRanges();
            window.getSelection().addRange(range);
            try {
                document.execCommand('copy');
                window.getSelection().removeAllRanges();
                Swal.fire({ title: 'คัดลอกสำเร็จ!', text: 'นำไปวางใน Excel หรือระบบ SGS ได้เลย', icon: 'success', timer: 1500, showConfirmButton: false });
            } catch (err) {
                Swal.fire('Error', 'ไม่สามารถคัดลอกได้', 'error');
            }
        }
